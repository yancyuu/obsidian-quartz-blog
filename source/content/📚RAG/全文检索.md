        cls,  
        query: str,  
        page: int = 1,  
        page_size: int = 10,  
        conditions: Optional[Dict[str, Any]] = None,  
        search_fields: Optional[List[str]] = None,  
        language: str = 'simple',  
        search_mode: str = 'any',

            f'WHERE {where_sql} '  
            f'ORDER BY {order_by} '  
            f'LIMIT {page_size} OFFSET {offset}'  
        )

        logger.info(f"查询参数: {params}")
